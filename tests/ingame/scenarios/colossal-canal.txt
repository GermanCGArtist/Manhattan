// colossal-canal: 44x3x7 canal trench + bridge schematic + flood + eat & defend triggers
stall(1)

// Areas
select(pos(100, -60, 100), pos(143, -58, 106))
save_area("colossal_trench")

// Places
let storage = pos(94, -57, 94)
let bridge_chests = pos(96, -57, 94)
let tools = pos(98, -57, 94)

let scaffold = "dirt"
#acceptableThrowawayItems {scaffold}

// Survival triggers
on food < 14
  eat()
  print("TEST colossal ate food={food}")
end

on mobs > 0
  defend()
  print("TEST colossal defended")
end

// ------------------------------------------------------------------ 1. Dig canal trench (44x3x7)
print("1/3: Digging colossal canal trench (44x3x7)")
area("colossal_trench")
let left_dig = clear_area(2)
if left_dig > 0
  #sel ca
end
dump(storage)
print("TEST trench dug left={left_dig}")

// ------------------------------------------------------------------ 2. Build bridge schematic across canal
print("2/3: Building bridge schematic across canal")
take_upto(bridge_chests, "stone_bricks", 64)
take_upto(bridge_chests, "cobblestone", 64)
take_upto(bridge_chests, "oak_stairs", 16)
take_upto(bridge_chests, "oak_slab", 32)
let left_bridge = schem_build("all", "add", "skip")
collect_scaffold()
schem_check("all", "add")
print("TEST bridge built left={left_bridge}")

// ------------------------------------------------------------------ 3. Flood canal with water
print("3/3: Flooding canal with water")
dump(storage)
take_upto(storage, scaffold, 128)
take_upto(tools, "bucket", 16)
try
  take_upto(tools, "water_bucket", 16)
catch err
  print("Water bucket fetch notice: {err}")
end
area("colossal_trench")
flood()
print("TEST colossal canal done")
